  const ZOOM_ICON = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M15 3h6v6M9 21H3v-6M21 3l-7 7M3 21l7-7"/></svg>';
  const CAM_ICON = '<svg viewBox="0 0 24 24" aria-hidden="true"><path d="M4 8h3l2-3h6l2 3h3v11H4z"/><circle cx="12" cy="13" r="3.5"/></svg>';
  const lb = document.getElementById('lb'), lbImg = document.getElementById('lb-img');
  const LB = { fotos: [], i: 0, cap: '' };
  const lbShow = () => {

    document.getElementById('lb-cap').textContent = LB.cap;
    const many = LB.fotos.length > 1;
    document.getElementById('lb-prev').hidden = !many; document.getElementById('lb-next').hidden = !many;
    document.getElementById('lb-n').textContent = many ? `${LB.i + 1} / ${LB.fotos.length}` : 'Tocá afuera o la × para cerrar';
  };
  const abrirFoto = (fotos, i, cap) => { LB.fotos = fotos; LB.i = i || 0; LB.cap = cap; lbShow(); if (lb.showModal) lb.showModal(); else lb.setAttribute('open', ''); };
  const step = (d) => { LB.i = (LB.i + d + LB.fotos.length) % LB.fotos.length; lbShow(); };
  const cerrar = () => { if (lb.close) lb.close(); else lb.removeAttribute('open'); };
  document.getElementById('lb-x').addEventListener('click', cerrar);
  document.getElementById('lb-prev').addEventListener('click', () => step(-1));
  document.getElementById('lb-next').addEventListener('click', () => step(1));
  document.getElementById('lb-view').addEventListener('click', (e) => { if (e.target !== lbImg) cerrar(); });
  lb.addEventListener('keydown', (e) => { if (e.key === 'ArrowLeft') step(-1); if (e.key === 'ArrowRight') step(1); });
  let tx = null;
  lb.addEventListener('touchstart', (e) => { tx = e.touches[0].clientX; }, { passive: true });
  lb.addEventListener('touchend', (e) => { if (tx == null || LB.fotos.length < 2) return; const dx = e.changedTouches[0].clientX - tx; if (Math.abs(dx) > 50) step(dx < 0 ? 1 : -1); tx = null; });

  const card = (p) => {
    const art = el('article', 'tag');
    const ph = el('div', 'ph' + (p.foto ? '' : ' none'));
    if (p.foto) {
      const img = el('img'); img.src = p.foto; img.alt = p.nombre; img.loading = 'lazy';
      img.addEventListener('click', () => abrirFoto([p.foto], 0, p.nombre));
      const z = el('button', 'zoom'); z.type = 'button'; z.innerHTML = ZOOM_ICON; z.append(document.createTextNode('Ver completa'));
      z.addEventListener('click', () => abrirFoto([p.foto], 0, p.nombre));
      ph.append(img, z);
    }
    else { ph.style.setProperty('--tint', tint(p.negocio)); ph.append(el('div', 'ini', p.nombre.split(' ').slice(0, 2).map((w) => w[0]).join('')), el('div', 'msg', 'Foto próximamente')); }
    if (p.ref && p.desde && p.ref > p.desde) { const bd = el('div', 'badge'); bd.setAttribute('aria-label', pct.format(p.ahorro) + (p.meli ? ' menos que en MercadoLibre' : ' de descuento')); bd.append(el('b', null, '−' + pct.format(p.ahorro)), el('small', null, p.meli ? 'vs MeLi' : 'Outlet')); ph.append(bd); }
    const nReal = p.unidades.filter((u) => u.fotos && u.fotos.length).length;
    if (nReal && p.foto) ph.append(el('span', 'hasreal', nReal === p.unidades.length ? '📷 Con foto real' : `📷 ${nReal} con foto real`));
    art.append(ph);
    const top = el('div', 'top');
    const c = el('span', 'cat', p.subcat || p.categoria || p.negocio); c.style.background = tint(p.negocio);
    top.append(c, el('h3', null, p.nombre));
    const dl = el('dl', 'ficha');
    for (const [k, v] of [['Rubro', [...new Set([p.negocio, p.categoria].filter(Boolean))].join(' · ')], ['Código', p.sku]]) { if (!v) continue; dl.append(el('dt', null, k), el('dd', null, v)); }
    top.append(dl);
    if (p.specs && p.specs.length) { const sp = el('ul', 'specs'); for (const s of p.specs) sp.append(el('li', null, s)); top.append(sp); }
    art.append(top);

    const pr = el('div', 'price');
    if (p.desde != null) {
      const conRef = p.ref && p.ref > p.desde;
      if (conRef) { const r = el('div', 'ref'); r.append(el('span', 'lbl', p.refLbl), el('span', 'was', ars.format(p.ref))); pr.append(r); }
      pr.append(el('span', 'now' + (conRef ? ' hot' : ''), (p.precios.filter(Boolean).some((x) => x.valor !== p.desde) ? 'desde ' : '') + ars.format(p.desde)));
      if (conRef) pr.append(el('span', 'save', 'Ahorrás ' + ars.format(p.ref - p.desde)));
    } else pr.append(el('span', 'now', 'Consultar'));
    pr.append(el('span', 'left' + (p.libres ? '' : ' allres'), !p.libres ? 'Reservado' : p.libres === 1 ? 'Queda 1' : `Quedan ${p.libres}`));
    art.append(pr);

    const ul = el('div', 'units');
    const abierto = state.abiertos.has(p.sku);
    const rank = (u) => (u.destino === 'RESERVADO' ? 2 : 0) + (u.fotos?.length ? 0 : 1);
    const orden = p.unidades.slice().sort((x, y) => rank(x) - rank(y));
    const visibles = abierto ? orden : orden.slice(0, 2);
    visibles.forEach((u) => {
      const i = p.unidades.indexOf(u), precio = p.precios[i];
      const res = u.destino === 'RESERVADO';
      const row = el('div', 'unit' + (res ? ' res' : ''));
      const d = el('div', 'd'); const cb = el('b', null, u.code); if (res) cb.append(el('span', 'rpill', 'Reservada')); d.append(cb, document.createTextNode(u.detalle || 'Sin detalle registrado: consultanos'));
      const pv = el('div', 'p', precio ? ars.format(precio.valor) : 'Consultar');
      if (precio && precio.rebaja) pv.append(el('small', null, '−10% por antigüedad'));
      const meliUrl = p.mla ? `https://articulo.mercadolibre.com.ar/${p.mla.replace('MLA', 'MLA-')}` : '';
      const msg = (res ? `Hola! Vi que la unidad ${u.code} de ${p.nombre} (${p.sku}) está reservada. Si se libera, me avisás?` : `Hola! Quiero comprar del Outlet: ${p.nombre} (${p.sku}), unidad ${u.code}${precio ? ', precio Outlet ' + ars.format(precio.valor) : ''}. ¿Sigue disponible?`) + (meliUrl ? `\n\nPublicación en MercadoLibre para comparar${p.meli ? ' (' + ars.format(p.meli) + ')' : ''}: ${meliUrl}` : '');
      const a = el('a', 'wa'); a.href = `https://wa.me/${WA}?text=${encodeURIComponent(msg)}`; a.target = '_blank'; a.rel = 'noopener'; a.innerHTML = WA_ICON; a.append(document.createTextNode(res ? 'Reservada · avisame si se libera' : 'Lo quiero por WhatsApp'));
      if (!res) a.addEventListener('click', () => reservar(p, u));
      row.append(d, pv);
      if (u.fotos && u.fotos.length) {
        const r = el('button', 'real'); r.type = 'button'; r.innerHTML = CAM_ICON;
        r.append(document.createTextNode('Ver foto real de esta unidad'));
        if (u.fotos.length > 1) r.append(el('small', null, `(${u.fotos.length})`));
        r.addEventListener('click', () => abrirFoto(u.fotos, 0, `${p.nombre} · ${u.code}${u.detalle ? ' · ' + u.detalle : ''}`));
        row.append(r);
      }
      row.append(a); ul.append(row);
    });
    if (p.unidades.length > 2) {
      const b = el('button', 'more', abierto ? 'Ver menos' : `Ver las ${p.unidades.length} unidades`); b.type = 'button';
      b.addEventListener('click', () => { abierto ? state.abiertos.delete(p.sku) : state.abiertos.add(p.sku); render(); });
      ul.append(b);
    }
    art.append(ul);
    if (p.mla) { const m = el('a', 'meli', 'Comparar con la publicación de MercadoLibre →'); m.href = `https://articulo.mercadolibre.com.ar/${p.mla.replace('MLA', 'MLA-')}`; m.target = '_blank'; m.rel = 'noopener'; art.append(m); }
    return art;
  };

  const render = () => {
    drawChips();
    let list = DATA.productos.filter((p) => (!state.negocio || p.negocio === state.negocio) && (!state.q || (p.nombre + ' ' + p.sku + ' ' + p.categoria + ' ' + p.subcat).toLowerCase().includes(state.q)));
    const by = { nuevo: (a, b) => b.ultimo.localeCompare(a.ultimo), 'precio-asc': (a, b) => (a.desde ?? 1e12) - (b.desde ?? 1e12), 'precio-desc': (a, b) => (b.desde ?? -1) - (a.desde ?? -1), ahorro: (a, b) => b.ahorro - a.ahorro };
    list = list.slice().sort(by[state.sort]);
    const grid = document.getElementById('grid'); grid.replaceChildren(...list.map(card));
    document.getElementById('empty').hidden = list.length > 0;
    const u = list.reduce((s, p) => s + p.unidades.length, 0);
    const libres = list.reduce((s, p) => s + p.libres, 0);
    document.getElementById('count').textContent = `${list.length} ${list.length === 1 ? 'producto' : 'productos'} · ${libres} ${libres === 1 ? 'unidad disponible' : 'unidades disponibles'}` + (VIVO ? ` · stock en vivo ${VIVO_HORA}` : ' · verificando stock…');

  const reservar = (p, u) => {
    try { fetch(`${ESTADO_URL}?reservar=${encodeURIComponent(u.code)}&sku=${encodeURIComponent(p.sku)}`, { mode: 'no-cors', keepalive: true }); } catch (e) {}
    VIVO = VIVO || {}; VIVO[u.code] = { sku: p.sku, destino: 'RESERVADO' };
    setTimeout(() => { preparar(); render(); }, 600);
    setTimeout(refrescar, 4000);
  };
  const refrescar = async () => {
    const ctl = new AbortController(); const to = setTimeout(() => ctl.abort(), 9000);
    try {
      const r = await fetch(ESTADO_URL, { signal: ctl.signal, cache: 'no-store' });
      const j = await r.json();
      if (!j || !j.estados) throw new Error('sin datos');
      VIVO = j.estados; VIVO_HORA = new Date().toLocaleTimeString('es-AR', { hour: '2-digit', minute: '2-digit' });
      preparar(); render();
    } catch (e) {
      const c = document.getElementById('count'); c.textContent = c.textContent.replace(' · verificando stock…', '');
    } finally { clearTimeout(to); }
  };
  document.addEventListener('visibilitychange', () => { if (document.visibilityState === 'visible') refrescar(); });
  render();
  refrescar();
